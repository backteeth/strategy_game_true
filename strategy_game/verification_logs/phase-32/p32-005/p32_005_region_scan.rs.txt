use strategy_game::hexmap::data::*;
fn dist(a:(i32,i32), b:(i32,i32)) -> i32 { let dq=a.0-b.0; let dr=a.1-b.1; (dq.abs()+dr.abs()+(dq+dr).abs())/2 }
fn main() {
    let m = load_from_path(DEFAULT_MAP_PATH).unwrap();
    let mut best: Vec<(i32,(i32,i32),usize,usize,usize,usize,usize)> = vec![];
    for c in m.cells() {
        if c.terrain.is_water() { continue; }
        let ctr = (c.q, c.r);
        let mut land=0; let mut water=0; let mut lake=0; let mut ring4land=0;
        for dq in -4..=4 { for dr in -4..=4 {
            let p=(c.q+dq,c.r+dr); if dist(p,ctr)>4 {continue;}
            if let Some(cell)=m.cell(p.0,p.1){ let d=dist(p,ctr);
                if cell.terrain==Terrain::Lake { lake+=1; }
                if d<=3 { if cell.terrain.is_water(){water+=1}else{land+=1} } else if !cell.terrain.is_water(){ring4land+=1}
            } else { lake+=100; } } }
        if lake>0 || land<22 || water<4 || ring4land<10 { continue; }
        let mut rivers=0;
        for e in m.river_edges() { let n=river_neighbor(e.q,e.r,e.direction); if dist((e.q,e.r),ctr)<=3 && dist(n,ctr)<=3 { let a=m.cell(e.q,e.r).map(|x|!x.terrain.is_water()).unwrap_or(false); let b=m.cell(n.0,n.1).map(|x|!x.terrain.is_water()).unwrap_or(false); if a&&b {rivers+=1} } }
        if rivers>=3 { best.push((rivers as i32,ctr,land,water,rivers,ring4land,0)); }
    }
    best.sort_by(|a,b| b.0.cmp(&a.0));
    for b in best.iter().take(12) { println!("center {:?} land {} water {} riverLandEdges {} ring4land {}", b.1,b.2,b.3,b.4,b.5); }
    println!("candidates {}", best.len());
}
